function openDraft(pos,box){

currentPosBox=box;

let source=
mode==="alltime"
?allTimeTeams
:currentTeams;

let rolled=
source[
Math.floor(Math.random()*source.length)
];

let players=
rolled.players[pos];

if(!players){
document.getElementById("draftInfo").innerHTML=
"No players available.";
return;
}

let html=`
<h2>${rolled.team}</h2>
`;

if(rolled.year){
html+=`<h3>${rolled.year}</h3>`;
}

html+=`<h3>${pos}</h3>`;

players.forEach(player=>{

if(
draftedPlayers.some(
p=>p.name===player.name
)
)return;

html+=`
<button class="rollBtn"
onclick="selectPlayer(
'${player.name}',
${player.rating}
)">
${player.name}
(${player.rating})
</button>
`;

});

document.getElementById("draftInfo").innerHTML=html;
}

function selectPlayer(name,rating){

draftedPlayers.some(
p=>p.name===name
)
)return;

draftedPlayers.push({

});

currentPosBox.innerHTML=
`<strong>${name}</strong><br>OVR ${rating}`;

`
<div class='playerCard'>
<h3>${name}</h3>
OVR ${rating}
</div>
`;
}

function updateOVR(){

if(draftedPlayers.length===0)return;

let total=0;

draftedPlayers.forEach(
p=>total+=p.rating
);

document.getElementById("ovr")
.innerText=
(total/draftedPlayers.length)
.toFixed(1);
}

function simulateSeason(){

if(draftedPlayers.length<13){

alert("Draft all 13 positions first!");

return;
}

let total=0;

draftedPlayers.forEach(
p=>total+=p.rating
);

let teamOVR=
total/13;

let wins=0;

let mvp=
draftedPlayers[
Math.floor(
Math.random()*draftedPlayers.length
)
];

for(let round=1;round<=27;round++){

let opponent=
85+
Math.random()*15;

let performance=
teamOVR+
(Math.random()*8-4);

if(performance>opponent){
wins++;
}

}

let losses=
27-wins;

let result="";

if(wins===27){

result=
"🏆 PERFECT SEASON 🏆\n\n"+
"27-0\n\n"+
"Greatest Team Ever Assembled\n\n";

}else{

result=
"Season Complete\n\n"+
wins+"-"+losses+"\n\n";

}

result+=
"MVP: "+mvp.name+
"\n\n"+
"Team OVR: "+
teamOVR.toFixed(1);

document.getElementById("result")
.innerText=result;
}
